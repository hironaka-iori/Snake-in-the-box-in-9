-- This notation expands to a list, so repeated corners remain separate positions.
local notation "pairCorners" => ([10,18,34,66,130,258,27,43,75,139,267,
  51,83,147,275,99,163,291,195,323,387,
  0,20,36,68,132,260,18,34,66,130,258,54,86,150,278,102,166,294,198,326,390] : List Nat)

theorem corner_list_eq :
    pairCorners = corners ⟨3,1,2,0,21⟩ ++ corners ⟨6,0,3,0,21⟩ := by
  decide +kernel

theorem corner_multiplicities :
    pairCorners.length = 42 ∧ pairCorners.toFinset.card = 37 ∧
    ([18,34,66,130,258] : List Nat).all
      (fun v => decide (pairCorners.count v = 2)) = true ∧
    pairCorners.all (fun v => decide
      (pairCorners.count v = if v ∈ ([18,34,66,130,258] : List Nat) then 2 else 1)) = true := by
  decide +kernel

theorem selected_absent_identity (S : Nat → Bool) :
    pairCorners.countP S + absentCorners S 3 1 2 + absentCorners S 6 0 3 = 42 := by
  have h2 := countP_corners_sel (S := S) (⟨3,1,2,0,21⟩ : Centre)
  have h4 := countP_corners_sel (S := S) (⟨6,0,3,0,21⟩ : Centre)
  have l2 : (freePairs 1 2).length = 21 := by decide +kernel
  have l4 : (freePairs 0 3).length = 21 := by decide +kernel
  change (corners ⟨3,1,2,0,21⟩).countP S + absentCorners S 3 1 2 =
    (freePairs 1 2).length at h2
  change (corners ⟨6,0,3,0,21⟩).countP S + absentCorners S 6 0 3 =
    (freePairs 0 3).length at h4
  rw [l2] at h2
  rw [l4] at h4
  rw [corner_list_eq, List.countP_append]
  omega

theorem actual_selected_bound {m : Nat} (p : Snake m) (P : CPerm) (r : Nat)
    (hr : r+7 ≤ m)
    (hshape : canon (dsOf p r 7) = [0,1,2,0,3,2,4])
    (hP : ∀ j, j < 7 → P.fwd (dir p (r+j)) =
      ([0,1,2,0,3,2,4] : List Nat).getD j 0) :
    pairCorners.countP (sel p (amap P (vx p r))) ≤ 34 := by
  have hp := (Snake9.ActualBounds125.prefix_and_suffix p r hr hshape).1
  have hpair := Snake9.ActualPair121.actual_pair_sum p r (by omega) hp
  have hc : Canon (winOf [0,1,2,0,3,2,4]) [0,1,2,0,3,2,4] := canon_winOf _
  have h2 := JointTags.frame_count p P r hc hr hP 1 (by decide)
  have h4 := JointTags.frame_count p P r hc hr hP 3 (by decide)
  have hw : winOf [0,1,2,0,3,2,4] = [0,1,3,7,6,14,10,26] := by decide +kernel
  have hd : dirOfStep 1 3 = 1 ∧ dirOfStep 3 7 = 2 ∧
      dirOfStep 7 6 = 0 ∧ dirOfStep 6 14 = 3 := by
    simp only [Tags.dirOfStep_eq]
    decide +kernel
  norm_num [hw, hd.1, hd.2.1, hd.2.2.1, hd.2.2.2, Nat.add_assoc] at h2 h4
  have h := selected_absent_identity (sel p (amap P (vx p r)))
  rw [h2,h4] at h
  omega

theorem actual_normalized_bound {m : Nat} (p : Snake m) (r : Nat)
    (hr : r+7 ≤ m)
    (hshape : canon (dsOf p r 7) = [0,1,2,0,3,2,4]) :
    ∃ P : CPerm,
      (∀ j, j < 7 → P.fwd (dir p (r+j)) =
        ([0,1,2,0,3,2,4] : List Nat).getD j 0) ∧
      pairCorners.countP (sel p (amap P (vx p r))) ≤ 34 := by
  obtain ⟨u, hu, hu9, hulen, hcs⟩ := actual_canonical_frame p r 7 hr
  have hsp := mkPerm_spec hu hu9 hulen
  have hP : ∀ j, j < 7 → (mkPerm u).fwd (dir p (r+j)) =
      ([0,1,2,0,3,2,4] : List Nat).getD j 0 := by
    intro j hj
    obtain ⟨hi,hd⟩ := hcs j hj
    rw [hshape] at hi hd
    rw [← hd]
    exact hsp _ hi
  exact ⟨mkPerm u,hP,actual_selected_bound p (mkPerm u) r hr hshape hP⟩

-- Only the literal-to-vertex encoding is a premise; the count bound is derived.
theorem actual_counter_good {m : Nat} (p : Snake m) (P : CPerm) (r : Nat)
    (hr : r+7 ≤ m)
    (hshape : canon (dsOf p r 7) = [0,1,2,0,3,2,4])
    (hP : ∀ j, j < 7 → P.fwd (dir p (r+j)) =
      ([0,1,2,0,3,2,4] : List Nat).getD j 0)
    (E : Nat → Bool) (M : MT) (ct : Ctr)
    (hpos : ct.lits.all (posP M) = true)
    (hmap : ct.lits.map (vl M) = pairCorners) (hk : ct.k = 34) :
    JointTags.Good (sel p (amap P (vx p r))) E M ct := by
  constructor
  · apply List.all_eq_true.2
    intro x hx
    have hp := List.all_eq_true.1 hpos x hx
    simp only [posP, Bool.and_eq_true, beq_iff_eq, decide_eq_true_eq] at hp
    simp [baseOK, hp.1.2, hp.2]
  · have hcount : ct.lits.countP
        (lv (base (sel p (amap P (vx p r))) E M)) =
        (ct.lits.map (vl M)).countP (sel p (amap P (vx p r))) := by
      rw [List.countP_map]
      apply List.countP_congr
      intro x hx
      simp [base_posP (List.all_eq_true.1 hpos x hx)]
    rw [hcount,hmap,hk]
    exact actual_selected_bound p P r hr hshape hP
