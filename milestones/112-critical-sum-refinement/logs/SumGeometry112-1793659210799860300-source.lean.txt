theorem actual_outer_sum
    (hlocal : ∀ S E : Nat → Bool,
      Axioms S E (winOf [0,1,2,3,1,4])
        (centresOf (winOf [0,1,2,3,1,4]) [(0,21),(0,21),(0,2),(0,21),(0,21)]) →
      absentCorners S 1 0 1 + absentCorners S 3 1 2 +
        absentCorners S 15 3 1 + absentCorners S 13 1 4 ≤ 18 → False)
    {m : Nat} (p : Snake m) (r : Nat) (hr : r+6 ≤ m)
    (hshape : canon (dsOf p r 6) = [0,1,2,3,1,4])
    (hmiddle : coreAlpha p (r+3) ≤ 2) :
    19 ≤ coreAlpha p (r+1)+coreAlpha p (r+2)+
      coreAlpha p (r+4)+coreAlpha p (r+5) := by
  obtain ⟨u, hu, hu9, hulen, hcs⟩ := actual_canonical_frame p r 6 hr
  have hsp := mkPerm_spec hu hu9 hulen
  have hP : ∀ j, j < 6 →
      (mkPerm u).fwd (dir p (r+j)) = ([0,1,2,3,1,4] : List Nat).getD j 0 := by
    intro j hj
    obtain ⟨hi, hd⟩ := hcs j hj
    rw [hshape] at hi hd
    rw [← hd]
    exact hsp _ hi
  have hc : Canon (winOf [0,1,2,3,1,4]) [0,1,2,3,1,4] := canon_winOf _
  have hb : ∀ i, i < ([(0,21),(0,21),(0,2),(0,21),(0,21)] : List (Nat × Nat)).length →
      (([(0,21),(0,21),(0,2),(0,21),(0,21)] : List (Nat × Nat)).getD i (0,21)).1 ≤
        coreAlpha p (r+i+1) ∧ coreAlpha p (r+i+1) ≤
      (([(0,21),(0,21),(0,2),(0,21),(0,21)] : List (Nat × Nat)).getD i (0,21)).2 := by
    intro i hi
    change i < 5 at hi
    have upper := coreAlpha_le_21 p (k := r+i+1) (by omega) (by omega)
    interval_cases i <;> simp_all [Nat.add_assoc]
  have ax := frame_axioms p (mkPerm u) r hc (by decide) hr hP hb
  have h1 := JointTags.frame_count p (mkPerm u) r hc hr hP 0 (by decide)
  have h2 := JointTags.frame_count p (mkPerm u) r hc hr hP 1 (by decide)
  have h4 := JointTags.frame_count p (mkPerm u) r hc hr hP 3 (by decide)
  have h5 := JointTags.frame_count p (mkPerm u) r hc hr hP 4 (by decide)
  have hw : winOf [0,1,2,3,1,4] = [0,1,3,7,15,13,29] := by decide +kernel
  norm_num [hw, Tags.dirOfStep_eq, Tags.lg, Nat.add_assoc] at h1 h2 h4 h5
  by_contra hsmall
  apply hlocal _ _ ax
  rw [h1,h2,h4,h5]
  omega
