+def build_gb(src_bytes):
+    m = src_bytes / 1e6
+    K = BUILD_GB_KNOTS

+    for (x0, y0), (x1, y1) in zip(K, K[1:]):
+        if m < x1:
+            return y0 + (y1 - y0) * (m - x0) / (x1 - x0)
